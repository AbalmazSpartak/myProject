import SwiftUI

struct FlashcardsView: View {
    @State private var currentIndex = 0
    @State private var showTranslation = false
    
    private let words = sampleWords
    
    var body: some View {
        VStack(spacing: 30) {
            Text("Изучение слов")
                .font(.largeTitle)
                .bold()
            
            Spacer()
            
            VStack(spacing: 15) {
                Text(words[currentIndex].english)
                    .font(.system(size: 40))
                    .bold()
                
                if showTranslation {
                    Text(words[currentIndex].russian)
                        .font(.title2)
                        .foregroundColor(.gray)
                } else {
                    Button("Показать перевод") {
                        showTranslation = true
                    }
                    .foregroundColor(.blue)
                }
            }
            .frame(maxWidth: .infinity)
            .padding(40)
            .background(Color.secondary.opacity(0.1))
            .cornerRadius(20)
            .padding(.horizontal)
            
            Spacer()
            
            Button(action: nextWord) {
                Text("Следующее слово")
                    .frame(maxWidth: .infinity)
                    .padding()
                    .background(Color.blue)
                    .foregroundColor(.white)
                    .cornerRadius(12)
            }
            .padding(.horizontal, 40)
        }
        .padding()
    }
    
    func nextWord() {
        showTranslation = false
        currentIndex = (currentIndex + 1) % words.count
    }
}

#Preview {
    FlashcardsView()
}
import Foundation

struct Word: Identifiable {
    let id = UUID()
    let english: String
    let russian: String
}

let sampleWords: [Word] = [
    Word(english: "Apple", russian: "Яблоко"),
    Word(english: "Book", russian: "Книга"),
    Word(english: "Cat", russian: "Кот"),
    Word(english: "Dog", russian: "Собака"),
    Word(english: "House", russian: "Дом")
]
